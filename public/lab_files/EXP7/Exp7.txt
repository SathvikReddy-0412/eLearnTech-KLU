/* USER CODE BEGIN 0 */
/* =========================================================
   LCD: Send 4 bits through PCF8574
   ========================================================= */
static void LCD_Write4Bits(uint8_t data)
{
    uint8_t temp;

    /* Backlight ON */
    temp = data | LCD_BACKLIGHT;
    HAL_I2C_Master_Transmit(&hi2c1, LCD_ADDR, &temp, 1, 100);

    /* Enable HIGH */
    temp = data | LCD_BACKLIGHT | LCD_ENABLE;
    HAL_I2C_Master_Transmit(&hi2c1, LCD_ADDR, &temp, 1, 100);

    HAL_Delay(1);

    /* Enable LOW */
    temp = data | LCD_BACKLIGHT;
    HAL_I2C_Master_Transmit(&hi2c1, LCD_ADDR, &temp, 1, 100);

    HAL_Delay(1);
}


/* =========================================================
   LCD: Send command/data byte
   rs = 0 -> command
   rs = 1 -> data
   ========================================================= */
static void LCD_Send(uint8_t data, uint8_t rs)
{
    uint8_t high;
    uint8_t low;

    high = data & 0xF0;
    low  = (data << 4) & 0xF0;

    LCD_Write4Bits(high | rs);
    LCD_Write4Bits(low | rs);
}


/* =========================================================
   LCD command
   ========================================================= */
void LCD_SendCommand(uint8_t command)
{
    LCD_Send(command, 0x00);

    if (command == 0x01 || command == 0x02)
        HAL_Delay(3);
    else
        HAL_Delay(1);
}


/* =========================================================
   LCD data
   ========================================================= */
void LCD_SendData(uint8_t data)
{
    LCD_Send(data, LCD_RS);
}


/* =========================================================
   LCD initialization
   ========================================================= */
void LCD_Init(void)
{
    HAL_Delay(100);

    /* Force LCD into 8-bit mode */
    LCD_Write4Bits(0x30);
    HAL_Delay(10);

    LCD_Write4Bits(0x30);
    HAL_Delay(5);

    LCD_Write4Bits(0x30);
    HAL_Delay(2);

    /* Switch to 4-bit mode */
    LCD_Write4Bits(0x20);
    HAL_Delay(2);

    /* 4-bit, 2 lines, 5x8 font */
    LCD_SendCommand(0x28);

    /* Display OFF */
    LCD_SendCommand(0x08);

    /* Clear display */
    LCD_SendCommand(0x01);

    /* Entry mode */
    LCD_SendCommand(0x06);

    /* Display ON, cursor OFF, blink OFF */
    LCD_SendCommand(0x0C);
}


/* =========================================================
   Set cursor
   ========================================================= */
void LCD_SetCursor(uint8_t row, uint8_t col)
{
    uint8_t address;

    if (row == 0)
        address = 0x80 + col;
    else
        address = 0xC0 + col;

    LCD_SendCommand(address);
}


/* =========================================================
   Print string
   ========================================================= */
void LCD_Print(char *str)
{
    while (*str)
    {
        LCD_SendData((uint8_t)*str);
        str++;
    }
}


/* =========================================================
   Print complete 16-character line
   Clears remaining characters
   ========================================================= */
void LCD_PrintLine(char *str)
{
    uint8_t i = 0;

    while (*str && i < 16)
    {
        LCD_SendData((uint8_t)*str);
        str++;
        i++;
    }

    while (i < 16)
    {
        LCD_SendData(' ');
        i++;
    }
}


/* =========================================================
   Clear LCD
   ========================================================= */
void LCD_Clear(void)
{
    LCD_SendCommand(0x01);
    HAL_Delay(3);

}
